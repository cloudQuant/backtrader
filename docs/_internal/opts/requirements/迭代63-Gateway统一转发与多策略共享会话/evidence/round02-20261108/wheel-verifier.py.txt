from bt_api_base.direct_write_protocol import DirectOrderReservation, DirectSubmitResult, DirectWriteErrorPolicy, NativeOrderIdentity, WriteOperation
from bt_api_gateway.direct_command_journal import GatewayDirectCommandJournal
from bt_api_gateway.session_guard import LocalAccountSessionGuard
from bt_api_py._contracts.models import OrderRequest, OrderType, Side
from bt_api_py.direct_write import DirectWriteCoordinator
from bt_api_py.gateway_direct_dispatch import compose_gateway_direct_dispatcher
from bt_api_py.gateway_direct_service import recovery_requested_operations
from bt_api_py.write_admission import WriteAdmission

route = 'SYNTHETIC'
account = 'synthetic-account'
guards = work / 'guards'
guards.mkdir(mode=0o700)
guard = LocalAccountSessionGuard(guards, account, 'synthetic-owner')
generation = guard.acquire()
now = [100.0]
dbfile = work / 'journal.sqlite'
journal = GatewayDirectCommandJournal.create(dbfile, account, generation=generation, clock=lambda: now[0])
class Provider:
    execution_ids_trading_day_scoped = True
    direct_write_error_policy = DirectWriteErrorPolicy(frozenset(), frozenset())
    def __init__(self): self.refs, self.sends = [], []
    def reserve_order(self, request):
        assert type(request) is OrderRequest
        ref = f'{len(self.refs) + 1:012d}'
        self.refs.append(ref)
        identity = NativeOrderIdentity(route, ref, {'trading_day': '20300101', 'front_id': 1, 'session_id': 7}, 1)
        return DirectOrderReservation(identity, request.symbol, ref)
    def release_reservation(self, reservation): pass
    def session_generation(self): return 1
    def submit_order(self, reservation, request, grant, verifier):
        with sqlite3.connect(dbfile) as db:
            status, payload = db.execute('SELECT status, payload_json FROM direct_journal_commands WHERE request_key=?', ('k',)).fetchone()
        assert status == 'DISPATCHING' and request.to_dict() == json.loads(payload)
        verifier.consume(grant, exchange_name=route, operation=WriteOperation.PLACE, session_generation=1)
        self.sends.append(reservation.identity.client_order_id)
        return DirectSubmitResult(True, reservation.identity)
provider = Provider()
coordinator = DirectWriteCoordinator(WriteAdmission({route: True}), lambda _: provider)
dispatcher = compose_gateway_direct_dispatcher(coordinator, route, journal, guard, lambda _: True, expected_local_provider=provider)
request = OrderRequest('X', Side.BUY, OrderType.LIMIT, Decimal('1.25'), account, 'cid', price=Decimal('120.50'))
row = journal.reserve('strategy', 'k', 'principal', request.to_dict())
observation = journal.observe('strategy', 'k', 'principal')
assert observation.record == row and observation.generation_current and observation.remaining_ttl_seconds == 10.0
now[0] = 104.0
assert journal.observe('strategy', 'k', 'principal').remaining_ttl_seconds == 6.0
args = (row.strategy_scope, row.request_key, row.principal, row.reservation_id, row.digest, row.version)
result = dispatcher.dispatch(*args)
assert result.status == 'SENT' and dispatcher.dispatch(*args) == result
assert provider.refs == provider.sends == ['000000000001']
assert journal.observe('strategy', 'k', 'principal').remaining_ttl_seconds is None
page = journal.recovery_page(limit=10)
assert len(page.records) == 1 and page.records[0] == result
assert recovery_requested_operations()['broker.direct_write'] == frozenset({'direct.command_status', 'direct.release'})
guard.release()

manifest = []
visited, unresolved, names_seen = set(), set(), set()

def copy_file(source, target):
    data = source.read_bytes()
    target.parent.mkdir(parents=True, exist_ok=True)
    target.write_bytes(data)
    manifest.append(f'{hashlib.sha256(data).hexdigest()} {source.relative_to(repo)} -> {target.relative_to(root)}')

for relative, source in sorted(mapping.items()):
    copy_file(repo / source, dest / relative)
# Existing joint test computes parents[1]/bt_api/bt_api_ctp/tests dynamically.

# Preserve unmodified source-path assertions in the CTP account-provider test:
# wheel target is .../install/src and the mirrored test is .../install/tests.
account_test = repo / 'bt_api/bt_api_ctp/tests/test_account_provider.py'
account_mirror = root.parent / 'install/tests/test_account_provider.py'
account_mirror.parent.mkdir(parents=True, exist_ok=True)
account_data = account_test.read_bytes()
account_mirror.write_bytes(account_data)
manifest.append(f'{hashlib.sha256(account_data).hexdigest()} {account_test.relative_to(repo)} -> install/tests/test_account_provider.py')
queue = list(dest.rglob('*.py')) + [root / 'bt_api/bt_api_ctp/tests/test_iter62_settlement_readback_on_write.py', account_mirror]
while queue:
    path = queue.pop()
    if path in visited: continue
    visited.add(path)
    tree = ast.parse(path.read_text(encoding='utf-8'), filename=str(path))
    names = set()
    for node in ast.walk(tree):
        if isinstance(node, ast.Import):
            names.update(alias.name for alias in node.names if alias.name.startswith(('tests.', 'test_')))
        elif isinstance(node, ast.ImportFrom) and node.level == 0 and node.module:
            if node.module.startswith(('tests.', 'test_')): names.add(node.module)
            elif node.module == 'tests': names.update('tests.' + alias.name for alias in node.names)
        elif isinstance(node, ast.Call) and node.args and isinstance(node.args[0], ast.Constant) and type(node.args[0].value) is str:
            value = node.args[0].value
            if isinstance(node.func, ast.Attribute) and node.func.attr == 'import_module' and value.startswith(('tests.', 'test_')):
                names.add(value)
    for name in sorted(names):
        names_seen.add(name)
        if name.startswith('tests.'):
            rel = Path(*name.split('.')[1:]).with_suffix('.py')
            source, target = repo / 'tests' / rel, dest / rel
        else:
            # Bare test_ helpers are CTP test fixtures; reject ambiguity rather than
            # falling through to an installed/source checkout import.
            candidates = [p for p in (repo / 'tests' / (name + '.py'), repo / 'bt_api/bt_api_ctp/tests' / (name + '.py')) if p.is_file()]
            if len(candidates) != 1:
                unresolved.add(name)
                continue
            source, target = candidates[0], dest / 'ctp' / (name + '.py')
        if not source.is_file():
            unresolved.add(name)
            continue
        if not target.exists(): copy_file(source, target)
        queue.append(target)
if unresolved:
    print('UNRESOLVED_TEST_IMPORTS', sorted(unresolved), flush=True)
    raise SystemExit(1)
(root / 'test-manifest.txt').write_text('\n'.join(sorted(set(manifest))) + '\n')
(root / 'test-imports.txt').write_text('\n'.join(sorted(names_seen)) + '\n')
print('copied public test/helper entries:', len(set(manifest)), 'reviewed:', len(visited), flush=True)

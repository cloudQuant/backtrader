from bt_api_base.direct_write_protocol import NativeOrderIdentity, PreDispatchRecord, WriteOperation
from bt_api_gateway.direct_command_journal import GatewayDirectCommandJournal
from bt_api_gateway.direct_dispatch import GatewayDirectDispatcher, NativeDispatchOutcome
from bt_api_gateway.session_guard import LocalAccountSessionGuard
from bt_api_py.gateway_direct_owner import GatewayDirectOwner, GatewayDirectOwnerResources, GatewayDispatchCommand

work = pathlib.Path(tempfile.mkdtemp(prefix='r04-wheel-smoke-')).resolve()
(work / 'guards').mkdir(mode=0o700)
shared, native_calls, fence_calls = {}, [], []
finalized = threading.Event()
def factory():
    guard = LocalAccountSessionGuard(work / 'guards', 'synthetic-account', 'owner')
    gen = guard.acquire()
    journal = GatewayDirectCommandJournal.create(work / 'journal.sqlite', 'synthetic-account', generation=gen)
    def native(payload, port, scope):
        identity = NativeOrderIdentity('TEST', 'ref1', {'trading_day': '20300101', 'front_id': 1}, 1)
        record = PreDispatchRecord(WriteOperation.PLACE, scope, 'TEST', 'X', identity, None, payload, '20300101', 1, 'synthetic')
        marker = port.commit(record)
        native_calls.append(marker)
        return NativeDispatchOutcome(True, identity, marker)
    dispatcher = GatewayDirectDispatcher(journal, 'TEST', guard, native, lambda _: ('strong-key',), lambda _: True)
    shared['journal'] = journal
    return GatewayDirectOwnerResources(dispatcher, journal, guard.release)
owner = GatewayDirectOwner(factory, queue_size=2, max_queue_wait_seconds=3, busy_timeout_seconds=3)
assert owner.start(3)
journal = shared['journal']
assert owner.is_bound_to(journal)
row = journal.reserve('strategy', 'key', 'principal', {'symbol': 'X', 'qty': '1'})
command = GatewayDispatchCommand(row.strategy_scope, row.request_key, row.principal, row.reservation_id, row.digest, row.version)
@contextmanager
def fence(row):
    fence_calls.append(('enter', row.request_key))
    yield lambda: fence_calls.append(('recheck', row.request_key))
    fence_calls.append(('exit', row.request_key))
assert owner.enqueue(command, request_fence=fence, finalize=finalized.set) == 'QUEUED'
assert finalized.wait(3)
result = journal.get('strategy', 'key', 'principal')
assert result.status == 'SENT' and native_calls == [row.reservation_id]
public = _public(journal.observe('strategy', 'key', 'principal'), journal)
assert public['record']['native_identity']['client_order_id'] == 'ref1'
assert public['record']['strong_identity_key_digests'] == [hashlib.sha256(b'strong-key').hexdigest()]
assert public['record']['reservation_generation'] == public['record']['owner_generation'] == row.generation
again = threading.Event()
assert owner.enqueue(command, request_fence=fence, finalize=again.set) == 'QUEUED'
assert again.wait(3)
assert native_calls == [row.reservation_id]
assert sum(kind == 'enter' for kind, _ in fence_calls) == 2
assert sum(kind == 'recheck' for kind, _ in fence_calls) == 1
assert owner.close(3) and not owner.status().thread_alive
print('OWNER_QUEUE_DISPATCH_FENCE_STATUS_DUPLICATE_CLEANUP passed')
